cmake_minimum_required(VERSION 3.16)
project(mini_nav_mapping_offline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(Eigen3 REQUIRED)
find_package(GTSAM REQUIRED)
find_package(octomap REQUIRED)
find_package(Threads REQUIRED)
add_executable(hba_offline ${HBA_ADAPTED_DIR}/hba_offline.cpp)
target_include_directories(hba_offline PRIVATE ${HBA_ADAPTED_DIR}/include ${PCL_INCLUDE_DIRS})
target_link_libraries(hba_offline PRIVATE ${PCL_LIBRARIES} gtsam Threads::Threads)
target_compile_options(hba_offline PRIVATE -Wall -Wextra -Wpedantic)
add_executable(replay_occupancy replay_occupancy.cpp)
target_include_directories(replay_occupancy PRIVATE ${PCL_INCLUDE_DIRS} ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(replay_occupancy PRIVATE ${PCL_LIBRARIES} ${OCTOMAP_LIBRARIES} Eigen3::Eigen)
target_compile_options(replay_occupancy PRIVATE -Wall -Wextra -Wpedantic)
