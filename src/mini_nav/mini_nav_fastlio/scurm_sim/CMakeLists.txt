cmake_minimum_required(VERSION 3.16)
project(scurm_sim)
find_package(ament_cmake REQUIRED)
get_filename_component(FASTLIO_ADAPTER
  "${CMAKE_CURRENT_SOURCE_DIR}/../../mini_nav_nodes/src/localization/fastlio2/adapter"
  ABSOLUTE)
install(DIRECTORY launch config models DESTINATION share/${PROJECT_NAME})
install(DIRECTORY maps DESTINATION share/${PROJECT_NAME} OPTIONAL)
install(PROGRAMS "${FASTLIO_ADAPTER}/adapter.py" DESTINATION lib/${PROJECT_NAME} RENAME adapter)
install(PROGRAMS "${FASTLIO_ADAPTER}/mapping_controls.py" DESTINATION lib/${PROJECT_NAME} RENAME mapping_controls)
install(PROGRAMS "${FASTLIO_ADAPTER}/mini_nav_localizer.py" DESTINATION lib/${PROJECT_NAME} RENAME mini_nav_localizer)
install(PROGRAMS "${FASTLIO_ADAPTER}/localization_backend.py" DESTINATION lib/${PROJECT_NAME} RENAME localization_backend)
install(FILES "${FASTLIO_ADAPTER}/localization_geometry.py" DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_adapter test/test_adapter.py)
  ament_add_pytest_test(test_localization_geometry test/test_localization_geometry.py)
  ament_add_pytest_test(test_initialpose test/test_initialpose.py)
endif()
ament_package()
