cmake_minimum_required(VERSION 3.16)
project(mini_nav_teleop)
find_package(ament_cmake REQUIRED)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rviz_common REQUIRED)
add_library(ps5_control_panel SHARED src/ps5_control_panel.cpp include/mini_nav_teleop/ps5_control_panel.hpp)
target_compile_features(ps5_control_panel PUBLIC cxx_std_17)
target_compile_options(ps5_control_panel PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(ps5_control_panel PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(ps5_control_panel rclcpp rcl_interfaces std_msgs std_srvs pluginlib rviz_common)
target_link_libraries(ps5_control_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins_description.xml)
install(TARGETS ps5_control_panel LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(PROGRAMS scripts/ps5_teleop.py DESTINATION lib/${PROJECT_NAME} RENAME ps5_teleop)
install(DIRECTORY vendor/dualforge DESTINATION lib/${PROJECT_NAME} PATTERN __pycache__ EXCLUDE)
install(DIRECTORY vendor/hid DESTINATION lib/${PROJECT_NAME} PATTERN __pycache__ EXCLUDE)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_index_cpp REQUIRED)
  find_package(geometry_msgs REQUIRED)
  ament_add_pytest_test(test_ps5_teleop test/test_ps5_teleop.py)
  ament_add_pytest_test(test_ps5_launch test/test_ps5_launch.py)
  set_tests_properties(test_ps5_launch PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=225")
  ament_add_gtest(test_ps5_panel test/test_ps5_panel.cpp SKIP_LINKING_MAIN_LIBRARIES TIMEOUT 60)
  target_compile_features(test_ps5_panel PRIVATE cxx_std_17)
  target_compile_definitions(test_ps5_panel PRIVATE PS5_TEST_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/test/ps5_usb_fixture.py")
  ament_target_dependencies(test_ps5_panel ament_index_cpp geometry_msgs rclcpp rcl_interfaces rviz_common std_msgs std_srvs)
  target_link_libraries(test_ps5_panel Qt5::Widgets)
  set_tests_properties(test_ps5_panel PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=222;QT_QPA_PLATFORM=xcb")
endif()
ament_package()
