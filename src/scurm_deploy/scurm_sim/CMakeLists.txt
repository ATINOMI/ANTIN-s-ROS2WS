cmake_minimum_required(VERSION 3.16)
project(scurm_sim)
find_package(ament_cmake REQUIRED)
install(DIRECTORY launch config models maps DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/adapter.py DESTINATION lib/${PROJECT_NAME} RENAME adapter)
install(PROGRAMS scripts/mapping_controls.py DESTINATION lib/${PROJECT_NAME} RENAME mapping_controls)
install(PROGRAMS scripts/mini_nav_localizer.py DESTINATION lib/${PROJECT_NAME} RENAME mini_nav_localizer)
install(PROGRAMS scripts/localization_backend.py DESTINATION lib/${PROJECT_NAME} RENAME localization_backend)
install(FILES scripts/localization_geometry.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_adapter test/test_adapter.py)
  ament_add_pytest_test(test_localization_geometry test/test_localization_geometry.py)
  ament_add_pytest_test(test_initialpose test/test_initialpose.py)
endif()
ament_package()
